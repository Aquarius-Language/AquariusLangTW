# The consumer programs of the native library tests (scripts/test-exports.ps1 configures this with a toolchain file
# from runtime/sdk/cmake/toolchains): C and C++ programs linking the test libraries the way any CMake program would,
# through each output's package config (lib/cmake/<name>/<name>-config.cmake):
#   AQUA_EXPORTS_GEOMETRY, AQUA_EXPORTS_COUNTER, AQUA_EXPORTS_SHAPES, AQUA_EXPORTS_SERVICES   <output>/<target> of each
#   AQUA_EXPORTS_KINDS        the kinds the libraries were built as (shared;static, and object where it exists): every
#                             consumer links the shared library; the C consumer and two_libraries also link each other kind
#   AQUA_EXPORTS_SANITIZE     build the consumers with AddressSanitizer (and LeakSanitizer) and UndefinedBehaviorSanitizer
cmake_minimum_required(VERSION 3.20)
project(aquarius_export_consumers C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AQUA_EXPORTS_SANITIZE "Build the consumers with sanitizers" OFF)
find_package(geometry CONFIG REQUIRED PATHS "${AQUA_EXPORTS_GEOMETRY}" NO_DEFAULT_PATH)
find_package(counter CONFIG REQUIRED PATHS "${AQUA_EXPORTS_COUNTER}" NO_DEFAULT_PATH)
find_package(shapes CONFIG REQUIRED PATHS "${AQUA_EXPORTS_SHAPES}" NO_DEFAULT_PATH)
find_package(services CONFIG REQUIRED PATHS "${AQUA_EXPORTS_SERVICES}" NO_DEFAULT_PATH)
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)

set(consumers "")
function(consumer name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE ${ARGN})
    set(consumers ${consumers} ${name} PARENT_SCOPE)
endfunction()
consumer(consumer_c consumer.c geometry::shared)
consumer(consumer_cpp consumer.cpp geometry::shared)
consumer(lifecycle lifecycle.c geometry::shared)
consumer(threads threads.c geometry::shared Threads::Threads)
consumer(two_libraries two_libraries.c geometry::shared counter::shared)
consumer(shapes shapes.c shapes::shared)
consumer(default_libraries services.c services::shared)
# The isolation tests: the same programs with each other kind; two Aquarius libraries in one program share no symbol.
foreach(kind IN LISTS AQUA_EXPORTS_KINDS)
    if(NOT kind STREQUAL "shared")
        consumer(consumer_c_${kind} consumer.c geometry::${kind})
        consumer(two_libraries_${kind} two_libraries.c geometry::${kind} counter::${kind})
    endif()
endforeach()
if(UNIX)
    consumer(reentrancy reentrancy.c counter::shared)
endif()
foreach(name IN LISTS consumers)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
    if(AQUA_EXPORTS_SANITIZE)
        target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
        target_link_options(${name} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
