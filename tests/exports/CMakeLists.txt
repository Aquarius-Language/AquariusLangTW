# The consumer programs of the native library tests (scripts/test-exports.ps1 configures this with a toolchain file
# from runtime/sdk/cmake/toolchains): C and C++ programs linking the geometry and counter libraries as any program
# would, from their output folders:
#   AQUA_EXPORTS_GEOMETRY, AQUA_EXPORTS_COUNTER   <output>/<target> of each library (include/, bin/ or lib/)
#   AQUA_EXPORTS_SANITIZE                          build the consumers with AddressSanitizer (and LeakSanitizer) and
#                                                  UndefinedBehaviorSanitizer (Linux)
cmake_minimum_required(VERSION 3.20)
project(aquarius_export_consumers C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AQUA_EXPORTS_SANITIZE "Build the consumers with sanitizers" OFF)

# Each library as an imported shared library: the DLL and its MinGW import library on Windows, the .so elsewhere.
foreach(name IN ITEMS geometry counter)
    string(TOUPPER "${name}" variable)
    set(folder "${AQUA_EXPORTS_${variable}}")
    add_library(${name} SHARED IMPORTED)
    set_target_properties(${name} PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${folder}/include")
    if(WIN32)
        set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${folder}/bin/${name}.dll" IMPORTED_IMPLIB "${folder}/lib/lib${name}.dll.a")
    else()
        set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${folder}/lib/lib${name}${CMAKE_SHARED_LIBRARY_SUFFIX}")
    endif()
endforeach()

set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
set(consumers consumer_c consumer_cpp lifecycle threads two_libraries)
add_executable(consumer_c consumer.c)
add_executable(consumer_cpp consumer.cpp)
add_executable(lifecycle lifecycle.c)
add_executable(threads threads.c)
add_executable(two_libraries two_libraries.c)
target_link_libraries(consumer_c PRIVATE geometry)
target_link_libraries(consumer_cpp PRIVATE geometry)
target_link_libraries(lifecycle PRIVATE geometry)
target_link_libraries(threads PRIVATE geometry Threads::Threads)
target_link_libraries(two_libraries PRIVATE geometry counter)
if(UNIX)
    list(APPEND consumers reentrancy)
    add_executable(reentrancy reentrancy.c)
    target_link_libraries(reentrancy PRIVATE counter)
endif()
foreach(consumer IN LISTS consumers)
    target_compile_options(${consumer} PRIVATE -Wall -Wextra -Werror)
    if(AQUA_EXPORTS_SANITIZE)
        target_compile_options(${consumer} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
        target_link_options(${consumer} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
