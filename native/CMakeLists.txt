cmake_minimum_required(VERSION 3.20)
# Link the CRT into our bridge and GLFW so standalone apps need no VC++ redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(USE_MSVC_RUNTIME_LIBRARY_DLL OFF CACHE BOOL "Use MSVC runtime library DLL" FORCE)
project(aquarius_graphics LANGUAGES C)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
include(FetchContent)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz
    URL_HASH SHA256=c038d34200234d071fae9345bc455e4a8f2f544ab60150765d7704e08f3dac01)
FetchContent_MakeAvailable(glfw)
if(WIN32)
    # Compile a guarded wrapper, preserving the pinned upstream source unchanged.
    get_target_property(aquarius_glfw_sources glfw SOURCES)
    set(aquarius_glfw_window_replaced FALSE)
    foreach(source IN LISTS aquarius_glfw_sources)
        if(source MATCHES "(^|/)win32_window\\.c$")
            list(REMOVE_ITEM aquarius_glfw_sources "${source}")
            set(aquarius_glfw_window_replaced TRUE)
        endif()
    endforeach()
    if(NOT aquarius_glfw_window_replaced)
        message(FATAL_ERROR "Cannot locate GLFW's Win32 window implementation")
    endif()
    list(APPEND aquarius_glfw_sources "${CMAKE_CURRENT_SOURCE_DIR}/glfw_win32_window.c")
    set_property(TARGET glfw PROPERTY SOURCES "${aquarius_glfw_sources}")
endif()
add_library(aquarius_graphics SHARED bridge.c text_input.c vendor/glad/src/gl.c)
target_include_directories(aquarius_graphics PRIVATE vendor/glad/include vendor/stb)
target_link_libraries(aquarius_graphics PRIVATE glfw)
if(WIN32)
    target_link_libraries(aquarius_graphics PRIVATE imm32)
endif()
if(UNIX AND NOT APPLE)
    if(GLFW_BUILD_X11)
        target_compile_definitions(aquarius_graphics PRIVATE AQUA_GLFW_X11)
    endif()
    if(GLFW_BUILD_WAYLAND)
        target_compile_definitions(aquarius_graphics PRIVATE AQUA_GLFW_WAYLAND)
    endif()
endif()
if(APPLE)
    enable_language(OBJC)
    target_sources(aquarius_graphics PRIVATE wgpu_surface.m)
    target_link_libraries(aquarius_graphics PRIVATE "-framework Cocoa" "-framework QuartzCore")
endif()
if(NOT WIN32)
    target_link_libraries(aquarius_graphics PRIVATE m ${CMAKE_DL_LIBS})
endif()
install(TARGETS aquarius_graphics RUNTIME DESTINATION . LIBRARY DESTINATION .)
add_library(aquarius_audio SHARED audio_output.c)
if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(aquarius_audio PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
install(TARGETS aquarius_audio RUNTIME DESTINATION . LIBRARY DESTINATION .)
