cmake_minimum_required(VERSION 3.22)
project(aquarius_android LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(wgpu_native SHARED IMPORTED)
set_target_properties(wgpu_native PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libwgpu_native.so")
add_library(aquarius_android SHARED gpu.cpp dynamic_library.cpp)
target_include_directories(aquarius_android PRIVATE include)
target_link_libraries(aquarius_android PRIVATE wgpu_native android log dl)
target_link_options(aquarius_android PRIVATE "-Wl,-z,max-page-size=16384")
